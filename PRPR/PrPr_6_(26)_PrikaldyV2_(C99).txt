/* ukazka1 2D staticke pole */

int main() {
	int i, j,c =1;
	int xa[M][N];
	int *p1, *p2;
	int  velkostVBytes;

	/* vypis - pristup cez [] */
	for (i = 0; i < M; i++)
		for (j = 0; j < N; j++)
			xa[i][j] = c++;

	for (i = 0; i < M; i++) {
		for (j = 0; j < N; j++)
			printf("%3d", xa[i][j]);
		printf("\n");
	}
	printf("\n ++++++++++++++++ \n \n"); /* vypis - pristup cez */

	for (i = 0; i < M; i++)
		for (j = 0; j < N; j++)
			*(*(xa+i)+j) = c++;

	for (i = 0; i < M; i++) {
		for (j = 0; j < N; j++)
			printf("%3d", *(*(xa + i) + j));
		printf("\n");
	}

	printf("\n ++++++++++++++++ \n \n"); /* vypis adries */
	for (i = 0; i < M; i++) {
		for (j = 0; j < N; j++)
			printf("%p  ", (*(xa + i) + j));
		printf("\n");
	}
	
	printf("\n ++++++++++++++++ \n \n"); /* vypis vzdialenosti v pameti medzi jednotlivymi riadkami - pozor vzdialenost je  v bytoch (1 byt = 8 bitov) */
	p1 = xa[0];
	for (i = 0; i < M; i++) {
		for (j = 0; j < N; j++) {
			p2 = (*(xa + i) + j);
			velkostVBytes = (p2 - p1) *sizeof(*p1);
			printf("%d  ", velkostVBytes);
			p1 = p2;
		}
		printf("\n");
	}

	p1 = &xa[0][0];
	p2 = &xa[1][0];
	velkostVBytes = (p2 - p1) * sizeof(*p1);
	printf("Rozdiel p2(A[0][0]) -p1(A[1][0])  %d  \n", velkostVBytes);
	p2 = &xa[0][1];
	velkostVBytes = (p2 - p1) * sizeof(*p1);
	printf("Rozdiel p2(A[0][0]) -p1(A[0][1])  %d (velkost int v bytoch = %d) \n", velkostVBytes,sizeof(int));

	return 0;
}

void vypis_2D_pole(int** p, int n, int m)
{
	int i,j;
	for ( i = 0; i < n; i++)
	{
		for (j = 0; j < m; j++)
			printf("%2d ", p[i][j]);
		printf("\n");
	}
	printf("\n");
}
int** alokuj(int n, int m)
{
	int i;
	int** vystup = (int**)malloc(n * sizeof(int*));
	for ( i = 0; i < n; i++)
		vystup[i] = (int *)malloc(m * sizeof(int));
	return vystup;
}
void uvolni(int** p, int n)
{
	int i;
	for (i = 0; i < n; i++)
		free(p[i]);
	free(p);
}
void napln2D(int*** p, int n, int m)
{
	int i,j;
	for ( i = 0; i < n; i++)

}

void nacitajtxt(FILE *fr, char** poleZnakov)
{
	char buf[50] = { 0 };
	int i = 0;
	rewind(fr);
	while (fgets(buf, 50, fr) != NULL) {
		//poleZnakov[i] = (char*)malloc(50 * sizeof(char));  // pole 50 znakov
		*(poleZnakov+i) = (char*)malloc(50 * sizeof(char));  // pole 50 znakov
		strcpy(*(poleZnakov + i), buf);
		i++;
	}
